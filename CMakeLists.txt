cmake_minimum_required(VERSION 3.20)
project(codex_meter VERSION 0.3.1 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(meter_options INTERFACE)
target_compile_definitions(meter_options INTERFACE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
if(MSVC)
  target_compile_options(meter_options INTERFACE /utf-8 /W4 /permissive-)
else()
  target_compile_options(meter_options INTERFACE -Wall -Wextra -Wno-missing-field-initializers -finput-charset=UTF-8 -fexec-charset=UTF-8)
  target_link_options(meter_options INTERFACE -static)
endif()
add_executable(codex_meter WIN32 src/main.cpp src/app.rc)
target_include_directories(codex_meter PRIVATE src assets)
target_link_libraries(codex_meter PRIVATE meter_options user32 gdi32 shell32 ole32 uuid comdlg32 comctl32 dwmapi gdiplus)
if(MINGW)
  target_link_options(codex_meter PRIVATE -municode)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(model_tests tests/model_tests.cpp)
  target_link_libraries(model_tests PRIVATE meter_options)
  add_executable(mock_codex tests/mock_codex.cpp)
  add_executable(render_tests tests/render_tests.cpp)
  target_link_libraries(render_tests PRIVATE meter_options gdi32 user32 shell32 ole32 uuid)
  target_link_libraries(mock_codex PRIVATE meter_options shell32 ole32 uuid)
  add_executable(rpc_tests tests/rpc_tests.cpp)
  target_link_libraries(rpc_tests PRIVATE meter_options user32 shell32 ole32 uuid)
  add_dependencies(rpc_tests mock_codex)
  add_test(NAME model COMMAND model_tests)
  add_test(NAME rendering COMMAND render_tests)
  add_test(NAME rpc_and_storage COMMAND rpc_tests)
endif()
install(TARGETS codex_meter RUNTIME DESTINATION .)
